# Optional: clear the screen on new terminal launch immediately
Clear-Host

Import-Module -Name Terminal-Icons
Import-Module PSReadLine
Import-Module PSFzf
Import-Module posh-git

Set-PSReadLineOption -PredictionSource History
Set-PSReadLineOption -PredictionViewStyle InlineView

# Navigation helpers
function .. { Set-Location .. }
function ... { Set-Location ../.. }

# Open current folder in explorer
function o { explorer.exe . }

# Git status shortcut function
function gs { git status }

# PSReadLine key bindings for history search
Set-PSReadLineKeyHandler -Key Ctrl+R -Function HistorySearchBackward
Set-PSReadLineKeyHandler -Key Ctrl+S -Function HistorySearchForward

# PSReadLine Catppuccin Mocha colors
Set-PSReadLineOption -Colors @{
    Command            = '#D9A6FF'
    Parameter          = '#F5C2E7'
    String             = '#ABE9B3'
    Operator           = '#F4DBD6'
    Variable           = '#F2CDCD'
    Comment            = '#6E6A86'
    Keyword            = '#F28FAD'
    Type               = '#96CDFB'
    Number             = '#F8BD96'
    Member             = '#89DCEB'
    Emphasis           = '#F2CDCD'
    Selection          = '#181825'
    Error              = '#F28FAD'
}

# Oh-My-Posh standalone CLI initialization
oh-my-posh init pwsh --config "$env:USERPROFILE\Documents\.poshthemes\catppuccin-mocha.omp.json" | Invoke-Expression

# Custom startup message
Write-Host "☕ Welcome back, $env:USERNAME! Today is $(Get-Date -Format 'dddd, MMMM dd yyyy, HH:mm')" -ForegroundColor Magenta
Write-Host "🚀 Have a productive session with Catppuccin Mocha vibes!" -ForegroundColor Green

# ASCII Art Injection
$asciiArtFile = "$env:USERPROFILE\Documents\6ix9ine_asciii.txt"
if (Test-Path $asciiArtFile) {
    Get-Content $asciiArtFile
}
